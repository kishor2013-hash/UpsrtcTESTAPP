async function handleDutySubmit(event) { event.preventDefault(); const user = getCurrentUser(); if (!user) return; const newDuty = { id: 'duty_' + Date.now(), userId: user.id, employeeId: user.employeeId, dutyDate: document.getElementById('duty-date').value, dutyNumber: document.getElementById('duty-num').value.trim(), shift: document.getElementById('duty-shift').value, busNumber: document.getElementById('duty-bus').value.trim(), routeFrom: document.getElementById('duty-from').value.trim(), routeTo: document.getElementById('duty-to').value.trim(), departureTime: document.getElementById('duty-dep-time').value, arrivalTime: document.getElementById('duty-arr-time').value, openingKm: parseFloat(document.getElementById('duty-open-km').value) || 0, closingKm: parseFloat(document.getElementById('duty-close-km').value) || 0, totalKm: parseFloat(document.getElementById('duty-total-km').value) || 0, passengerCount: parseInt(document.getElementById('duty-passengers').value) || 0, totalSeats: parseInt(document.getElementById('duty-seats').value) || 52, loadFactor: parseFloat(document.getElementById('duty-load-factor').value) || 0, income: parseFloat(document.getElementById('duty-income').value) || 0, expenses: parseFloat(document.getElementById('duty-expenses').value) || 0, remarks: document.getElementById('duty-remarks').value.trim() }; // Immediate local cache for offline resiliency const duties = JSON.parse(localStorage.getItem(STORAGE_KEY_DUTIES) || '[]'); duties.push(newDuty); localStorage.setItem(STORAGE_KEY_DUTIES, JSON.stringify(duties)); // Send to Hostinger MySQL backend via Fetch API POST try { const response = await fetch('save_data.php', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' }, body: JSON.stringify({ action: 'save_duty', ...newDuty }) }); if (response.ok) { const result = await response.json(); if (result.status === 'success') { showToast(`Duty Saved to Hostinger MySQL! (Bus: ${newDuty.busNumber})`); } else { showToast(`Saved locally. (${result.message})`); } } else { showToast(`Saved locally. (Hostinger HTTP ${response.status})`); } } catch (err) { showToast('Saved to Local Cache (Hostinger sync ready)'); } finally { navigateTo('screen-show-data'); } }
